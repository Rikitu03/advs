# ADVS ML API — inference-only dependencies for the FastAPI service (python/api).

# Training dependencies stay in requirements.txt; this file is what the
# Dockerfile installs so the deployed image carries no trainers, tuners, or
# synthetic-data generators. Local dev installs this INTO the existing venv
# (python/env/Scripts/python.exe -m pip install -r requirements-api.txt).

# --- web service ---
fastapi>=0.115
uvicorn[standard]>=0.30
python-multipart>=0.0.9       # multipart file uploads
pydantic-settings>=2.4        # env-driven Settings (model paths, thresholds)
httpx>=0.27                   # fastapi TestClient transport (tests + health probes)

# --- numerics / imaging (match requirements.txt pins) ---
numpy>=1.26,<2.1
pillow>=10.0
opencv-python-headless>=4.9

# --- Stage 2: OCR ---
pytesseract>=0.3.10           # needs the tesseract binary (apt: tesseract-ocr)

# --- Stage 3: ResNet-50 classifier (.keras / .h5) ---
# tensorflow-cpu: the service is CPU-only on HF free tier; 2.16 is the first
# release with Python 3.12 wheels (see requirements.txt note).

# --- Stage 4: YOLOv8 detector (pulls in torch/torchvision) ---

ultralytics>=8.1

# --- Stage T: forensics (metadata reads) ---
pikepdf>=8.0

# --- Stage 2: ROI+TrOCR field-recognition fallback (roi_field_ocr.py) ---
# rapidocr-onnxruntime: PP-OCRv4 text DETECTOR only (recognition unused, TrOCR
# replaces it) — pulls onnxruntime, which (unlike onnxruntime-gpu) ships
# official CPU wheels for both x86_64 and aarch64 (Oracle Ampere A1 target).
rapidocr-onnxruntime>=1.4
# <5: transformers 5.x's rewritten tokenizer loader (TokenizersBackend) fails
# to load microsoft/trocr-large-printed's tokenizer ("need sentencepiece or
# tiktoken" even with both installed) - confirmed against 5.14.1; 4.57.6 works.
transformers>=4.40,<5          # microsoft/trocr-large-printed (VisionEncoderDecoderModel)
sentencepiece>=0.2             # TrOCR's tokenizer
